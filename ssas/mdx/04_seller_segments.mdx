-- Requires load_seller_segments.sql, updated SSAS source and Process Full.
-- Labels KMeans C0..C4 are candidates, not final business segment names.
-- Slice: sales of the KMeans C4 candidate.
SELECT {[Measures].[Price], [Measures].[Freight Value]} ON COLUMNS,
 NON EMPTY [Dim Location].[Geolocation State].[Geolocation State].MEMBERS ON ROWS
FROM [Olist DW]
WHERE ([Dim Seller].[Seller Segment].&[KMeans C4]);

-- Pivot: Segment x state, displaying price.
SELECT NON EMPTY [Dim Seller].[Seller Segment].[Seller Segment].MEMBERS ON COLUMNS,
 NON EMPTY [Dim Location].[Geolocation State].[Geolocation State].MEMBERS ON ROWS
FROM [Olist DW]
WHERE ([Measures].[Price]);

-- Region of seller: not the same as the location dimension's order/customer geography.
SELECT {[Measures].[Price]} ON COLUMNS,
 NON EMPTY CROSSJOIN(
  [Dim Seller].[Seller Region].[Seller Region].MEMBERS,
  [Dim Seller].[Seller Segment].[Seller Segment].MEMBERS) ON ROWS
FROM [Olist DW];
